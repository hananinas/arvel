cmake_minimum_required(VERSION 3.16)
project(arvel_ui_behavior C)
get_filename_component(repo "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_subdirectory("${repo}/components/lvgl" lvgl EXCLUDE_FROM_ALL)
target_compile_definitions(lvgl PUBLIC LV_CONF_SKIP=1 LV_MEM_CUSTOM=1 LV_COLOR_DEPTH=16 LV_USE_PNG=1 LV_FONT_MONTSERRAT_12=1 LV_FONT_MONTSERRAT_14=1 LV_FONT_MONTSERRAT_16=1)
file(GLOB fonts "${repo}/components/ui_bsp/fonts/*.c")
add_executable(ui_behavior behavior.c "${repo}/components/ui_bsp/buddy_ui.c" "${repo}/components/ui_bsp/arvel_art.c" ${fonts})
target_include_directories(ui_behavior PRIVATE "${repo}/components/ui_bsp" "${repo}/components/lvgl/src")
target_link_libraries(ui_behavior PRIVATE lvgl m)
